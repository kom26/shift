# シフト管理 iOS v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `index.html` のシフト管理ツールのコア機能(月/週表示、予定の追加・編集・複製・削除、色とフィルタ、祝日、勤務時間計算、集計、JSON取り込み)を SwiftUI の iOS アプリとして実装する。

**Architecture:** UI非依存の純粋ロジック(`Core/`)と SwiftUI ビュー(`Views/`)を分ける。予定は `[ShiftEvent]` を1つのJSONファイルに保存し、`@Observable` の `Store` が唯一の状態源になる。複数日バーの配置計算(`SpanLayout`)は月/週ビューで共用する。

**Tech Stack:** Swift 5 モード、SwiftUI、Observation、Swift Testing、xcodegen、Codemagic(CI)。外部ライブラリなし。

## Global Constraints

設計書 `docs/superpowers/specs/2026-10-24-ios-swiftui-design.md` の内容(Task 1 で下記の修正を反映する)。

- 対象は iOS 17+、SwiftUI、外部ライブラリなし
- Xcodeプロジェクトは xcodegen(`ios/project.yml`)で生成し、`.xcodeproj` はコミットしない
- 日付は `"yyyy-MM-dd"`、時刻は `"HH:mm"`(終了は `"24:00"` を許容)の文字列で保持する(Web版と同一挙動)
- 項目追加はOptionalで行い、スキーマ移行を不要にする
- 休憩の既定値は、総時間が6時間超で45分、8時間超で60分
- 設定は `@AppStorage`、色フィルタ・最近使った色は UserDefaults、予定はJSONファイル
- v1に含めない: ドラッグ移動、リサイズ、未配置エリア、一括インポート、ウィジェット、JSON書き出し(v2/v3)

## 検証の方式(重要)

作業PCはWindowsで Swift ツールチェーンがない。コンパイルとテストは Codemagic(macOS)でのみ実行できる。

- テストは実装より先に書くが、赤→緑の確認は CI チェックポイントでまとめて行う
- **チェックポイントは Task 1 / 4 / 9**。それ以外のタスクはローカルコミットのみ
- `git push` はリポジトリ外への公開にあたるため、**初回 push の前にユーザーへ確認する**
- CI が失敗したら、ログの最初のエラーだけを直して再 push する。複数のタスクをまたぐ失敗は、該当タスクのファイルだけを直す
- 前提: ユーザーが Codemagic に `kom26/shift` を接続し、リポジトリ直下の `codemagic.yaml` を使う設定にしていること(Task 1 の Step 8)

`ios/Shift/Core/ShiftEvent.swift`:

```swift
import Foundation

struct ShiftEvent: Codable, Identifiable, Equatable {
    var id: String
    var title: String
    var memo: String
    var color: String
    var startDate: String?
    var endDate: String?
    var placed: Bool
    var allDay: Bool
    var startTime: String?
    var endTime: String?
    var workHours: Double?

    enum CodingKeys: String, CodingKey {
        case id, title, memo, color, startDate, endDate, placed, allDay, startTime, endTime, workHours
    }

    static func new(on date: String) -> ShiftEvent {
        ShiftEvent(id: UUID().uuidString, title: "", memo: "", color: "#6366f1",
                   startDate: date, endDate: date, placed: true, allDay: true,
                   startTime: nil, endTime: nil, workHours: nil)
    }

    var effectiveHours: Double? {
        workHours ?? Rules.workHours(start: startTime, end: endTime, allDay: allDay)
    }
}

extension ShiftEvent {
    /// Web版のJSONは項目が欠けていることがあるため、欠落は既定値で補う
    init(from decoder: Decoder) throws {
        let c = try decoder.container(keyedBy: CodingKeys.self)
        id = try c.decodeIfPresent(String.self, forKey: .id) ?? UUID().uuidString
        title = try c.decodeIfPresent(String.self, forKey: .title) ?? ""
        memo = try c.decodeIfPresent(String.self, forKey: .memo) ?? ""
        color = try c.decodeIfPresent(String.self, forKey: .color) ?? "#6366f1"
        startDate = try c.decodeIfPresent(String.self, forKey: .startDate).nonEmpty
        endDate = try c.decodeIfPresent(String.self, forKey: .endDate).nonEmpty
        placed = try c.decodeIfPresent(Bool.self, forKey: .placed) ?? (startDate != nil)
        allDay = try c.decodeIfPresent(Bool.self, forKey: .allDay) ?? true
        startTime = try c.decodeIfPresent(String.self, forKey: .startTime).nonEmpty
        endTime = try c.decodeIfPresent(String.self, forKey: .endTime).nonEmpty
```

```swift
import Foundation

enum DayType { case normal, saturday, sunday, holiday }

/// 日付は "yyyy-MM-dd" 文字列で扱い、計算時だけ正午のDateに変換する(夏時間や時差の影響を避けるため)
enum Rules {
    private static let cal = Calendar(identifier: .gregorian)

    static let timeOptions: [String] = (0...48).map { String(format: "%02d:%@", $0 / 2, $0 % 2 == 0 ? "00" : "30") }

    // MARK: 日付

    static func ymd(_ s: String) -> (year: Int, month: Int, day: Int) {
        let p = s.split(separator: "-").compactMap { Int($0) }
        return p.count == 3 ? (p[0], p[1], p[2]) : (1970, 1, 1)
    }

    static func parse(_ s: String) -> Date? {
        let p = s.split(separator: "-").compactMap { Int($0) }
        guard p.count == 3 else { return nil }
        return cal.date(from: DateComponents(year: p[0], month: p[1], day: p[2], hour: 12))
    }

    static func format(_ d: Date) -> String {
        let c = cal.dateComponents([.year, .month, .day], from: d)
        return String(format: "%04d-%02d-%02d", c.year!, c.month!, c.day!)
    }

    static func today() -> String { format(Date()) }

    /// 0 = 日曜 … 6 = 土曜(JSの getDay() と同じ)
    static func weekday(_ s: String) -> Int {
        guard let d = parse(s) else { return 0 }
        return cal.component(.weekday, from: d) - 1
    }

    static func addDays(_ s: String, _ n: Int) -> String {
        guard let d = parse(s), let r = cal.date(byAdding: .day, value: n, to: d) else { return s }
        return format(r)
    }

    static func addMonths(_ s: String, _ n: Int) -> String {
        guard let d = parse(s), let r = cal.date(byAdding: .month, value: n, to: d) else { return s }
        return format(r)
    }

    static func weekStart(of s: String, startMonday: Bool) -> String {
        let day = weekday(s)
        let offset = startMonday ? (day == 0 ? -6 : 1 - day) : -day
        return addDays(s, offset)
    }

    static func monthWeeks(year: Int, month: Int, startMonday: Bool) -> [[String?]] {
        let first = String(format: "%04d-%02d-01", year, month)
        let w = weekday(first)
        let offset = startMonday ? (w + 6) % 7 : w
        let days = cal.range(of: .day, in: .month, for: parse(first)!)!.count
        var cells: [String?] = Array(repeating: nil, count: offset)
        for d in 1...days { cells.append(String(format: "%04d-%02d-%02d", year, month, d)) }
        while cells.count % 7 != 0 { cells.append(nil) }
        return stride(from: 0, to: cells.count, by: 7).map { Array(cells[$0..<$0 + 7]) }
    }

    static func monthBounds(of s: String) -> (first: String, last: String) {
        let (y, m, _) = ymd(s)
        let first = String(format: "%04d-%02d-01", y, m)
        let days = cal.range(of: .day, in: .month, for: parse(first)!)!.count
        return (first, String(format: "%04d-%02d-%02d", y, m, days))
    }

    static func weekdayLabels(startMonday: Bool) -> [String] {
        let sun = ["日", "月", "火", "水", "木", "金", "土"]
        return startMonday ? Array(sun[1...]) + [sun[0]] : sun
    }

    static func dateInRange(_ ds: String, _ start: String, _ end: String?) -> Bool {
        ds >= start && ds <= (end ?? start)
    }

    static func dayType(_ ds: String, isHoliday: Bool) -> DayType {
        if isHoliday { return .holiday }
        switch weekday(ds) {
        case 0: return .sunday
        case 6: return .saturday
        default: return .normal
        }
    }

    // MARK: 時間

    static func resolveEndDate(start: String?, end: String?, allDay: Bool, startTime: String?, endTime: String?) -> String? {
        guard let sd = start.nonEmpty else { return end }
        let ed = end.nonEmpty
        guard !allDay, let st = startTime.nonEmpty, let et = endTime.nonEmpty else { return ed ?? sd }
        if et == "24:00" { return ed ?? sd }
        if et <= st { return addDays(sd, 1) }
        if let ed, ed >= sd { return ed }
        return sd
    }

    static func minutes(_ t: String?) -> Int? {
        guard let t = t.nonEmpty else { return nil }
        let p = t.split(separator: ":").compactMap { Int($0) }
        guard p.count == 2 else { return nil }
        return p[0] * 60 + p[1]
    }

    static func totalMinutes(start: String?, end: String?) -> Int? {
        guard let s = minutes(start), let e = minutes(end) else { return nil }
        var m = e - s
        if m <= 0 { m += 1440 }
        return m
    }

    static func defaultBreak(_ total: Int) -> Int {
        total > 480 ? 60 : total > 360 ? 45 : 0
    }

    static func workHours(start: String?, end: String?, allDay: Bool, breakMinutes: Int? = nil) -> Double? {
        guard !allDay, let total = totalMinutes(start: start, end: end) else { return nil }
        let brk = breakMinutes ?? defaultBreak(total)
        return (Double(total - brk) / 60 * 100).rounded() / 100
    }

    /// 保存済みの workHours から、既定値と異なる休憩分を逆算する
    static func breakOverride(of ev: ShiftEvent) -> Int? {
        guard let wh = ev.workHours, !ev.allDay,
              let total = totalMinutes(start: ev.startTime, end: ev.endTime) else { return nil }
        let brk = Int((Double(total) - wh * 60).rounded())
        return brk != defaultBreak(total) ? brk : nil
    }

    static func formatHours(_ h: Double?) -> String {
        guard let h, !h.isNaN else { return "-" }
        let hh = Int(h.rounded(.down))
        let mm = Int(((h - Double(hh)) * 60).rounded())
        return mm > 0 ? "\(hh)h\(mm)m" : "\(hh)h"
    }

    static func timeLabel(_ ev: ShiftEvent) -> String {
        if ev.allDay { return "終日" }
        if let s = ev.startTime, let e = ev.endTime { return "\(s)〜\(e)" }
        if let s = ev.startTime { return "\(s)〜" }
        return ""
    }

    /// 週ビューの1日分の縦位置(0...1)。日またぎの予定は開始日・終了日・中間日で形が変わる
    static func timedSegment(_ ev: ShiftEvent, on ds: String) -> (top: Double, height: Double)? {
        guard !ev.allDay, let start = ev.startDate, dateInRange(ds, start, ev.endDate) else { return nil }
        let end = ev.endDate ?? start
        let isStart = start == ds, isEnd = end == ds
        let s = Double(minutes(ev.startTime) ?? 0)
        let e = Double(minutes(ev.endTime) ?? 1440)
        if isStart && isEnd {
            let e2 = e <= s ? 1440 : e
            return (s / 1440, max(30, e2 - s) / 1440)
        } else if isStart {
            return (s / 1440, (1440 - s) / 1440)
        } else if isEnd {
            return (0, max(30, e) / 1440)
        }
        return (0, 1)
    }
}
```

```swift
import Foundation

struct BarSpan: Identifiable, Equatable {
    let event: ShiftEvent
    let col: Int
    let span: Int
    let isStart: Bool
    let isEnd: Bool
    var id: String { event.id }
}

enum SpanLayout {
    /// 1週間分の予定を、重ならないバーの行に詰める(Web版 computeWeekSpans 相当)
    static func rows(events: [ShiftEvent], weekCells: [String?]) -> [[BarSpan]] {
        let valid = weekCells.compactMap { $0 }
        guard let first = valid.first, let last = valid.last else { return [] }

        var spans: [BarSpan] = []
        for ev in events where ev.placed {
            guard let start = ev.startDate else { continue }
            let end = ev.endDate ?? start
            if start > last || end < first { continue }
            guard let cs = weekCells.firstIndex(where: { $0.map { $0 >= start } ?? false }) else { continue }
            let ce = weekCells.lastIndex(where: { $0.map { $0 <= end } ?? false }) ?? 6
            guard ce >= cs else { continue }
            spans.append(BarSpan(event: ev, col: cs, span: ce - cs + 1,
                                 isStart: start >= first && start <= last,
                                 isEnd: end >= first && end <= last))
        }

        // 同じ列は入力順を保つ(sort は安定ではないため添字で比較する)
        let sorted = spans.enumerated()
            .sorted { ($0.element.col, $0.offset) < ($1.element.col, $1.offset) }
            .map(\.element)

        var rows: [[BarSpan]] = []
        for s in sorted {
            let free = rows.firstIndex { row in
                !row.contains { $0.col < s.col + s.span && $0.col + $0.span > s.col }
            }
            if let free { rows[free].append(s) } else { rows.append([s]) }
        }
        return rows
    }
}
```

```swift
import Foundation
import Observation

@Observable
final class Holidays {
    private(set) var table: [String: String]
    @ObservationIgnored private var fetched: Set<String>
    @ObservationIgnored private let defaults: UserDefaults
    private static let key = "shift-holidays"

    init(defaults: UserDefaults = .standard) {
        self.defaults = defaults
        var t = BuiltinHolidays.table
        if let data = defaults.data(forKey: Self.key),
           let cached = try? JSONDecoder().decode([String: String].self, from: data) {
            t.merge(cached) { _, new in new }
        }
        table = t
        fetched = Set(t.keys.map { String($0.prefix(4)) })
    }

    func name(on ds: String) -> String? { table[ds] }

    /// 前年・当年・翌年を未取得なら holidays-jp から取得する。失敗しても内蔵テーブルで動く
    func load(around year: Int) async {
        for y in [year - 1, year, year + 1] {
            let k = String(y)
            guard !fetched.contains(k) else { continue }
            fetched.insert(k)
            guard let url = URL(string: "https://holidays-jp.github.io/api/v1/\(k)/date.json"),
                  let (data, res) = try? await URLSession.shared.data(from: url),
                  (res as? HTTPURLResponse)?.statusCode == 200,
                  let dict = try? JSONDecoder().decode([String: String].self, from: data) else { continue }
            table.merge(dict) { _, new in new }
            if let encoded = try? JSONEncoder().encode(table) { defaults.set(encoded, forKey: Self.key) }
        }
    }
```

```swift
import Foundation
import Observation

struct ColorTotal: Identifiable {
    let color: String
    var hours: Double
    var count: Int
    var id: String { color }
}

@Observable
final class Store {
    static let defaultColors = ["#6366f1", "#06b6d4", "#f59e0b", "#ec4899", "#10b981", "#f97316", "#8b5cf6", "#ef4444"]

    static var defaultFileURL: URL {
        let dir = URL.applicationSupportDirectory
        try? FileManager.default.createDirectory(at: dir, withIntermediateDirectories: true)
        return dir.appending(path: "events.json")
    }

    private(set) var events: [ShiftEvent] = []
    private(set) var checkedColors: Set<String>
    private(set) var customColors: [String]

    @ObservationIgnored private let fileURL: URL
    @ObservationIgnored private let defaults: UserDefaults
    private enum Keys {
        static let checked = "shift-checked-colors"
        static let custom = "shift-custom-colors"
    }

    init(fileURL: URL = Store.defaultFileURL, defaults: UserDefaults = .standard) {
        self.fileURL = fileURL
        self.defaults = defaults
        checkedColors = Set(defaults.stringArray(forKey: Keys.checked) ?? [])
        customColors = defaults.stringArray(forKey: Keys.custom) ?? []
        if let data = try? Data(contentsOf: fileURL),
           let list = try? JSONDecoder().decode([ShiftEvent].self, from: data) {
            events = list
        }
    }

    // MARK: 参照

    /// 配置済みで、色フィルタを通る予定(フィルタが空なら全色)
    var visibleEvents: [ShiftEvent] {
        events.filter { $0.placed && $0.startDate != nil && (checkedColors.isEmpty || checkedColors.contains($0.color)) }
    }

    var colorsInUse: [String] {
        var seen = Set<String>()
        return events.map(\.color).filter { seen.insert($0).inserted }
    }

    /// 開始日が from...to に入る予定を色ごとに合計する
    func totals(from: String, to: String) -> [ColorTotal] {
        var order: [String] = []
        var map: [String: ColorTotal] = [:]
        for ev in visibleEvents {
            guard let sd = ev.startDate, sd >= from, sd <= to,
                  let h = ev.effectiveHours, h > 0 else { continue }
            if map[ev.color] == nil {
                order.append(ev.color)
                map[ev.color] = ColorTotal(color: ev.color, hours: 0, count: 0)
            }
            map[ev.color]!.hours += h
            map[ev.color]!.count += 1
        }
        return order.compactMap { map[$0] }
    }

    // MARK: 変更

    func save(_ event: ShiftEvent) {
        if let i = events.firstIndex(where: { $0.id == event.id }) {
            events[i] = event
        } else {
            events.append(event)
        }
        persist()
    }

    /// 編集内容を保存したうえで、同じ内容のコピーを別IDで追加する(Web版の「複製保存」)
    func saveAndDuplicate(_ event: ShiftEvent) {
        save(event)
        var copy = event
        copy.id = UUID().uuidString
        events.append(copy)
        persist()
    }

    func delete(id: String) {
        events.removeAll { $0.id == id }
        persist()
    }

    func toggleColor(_ color: String) {
        if checkedColors.contains(color) { checkedColors.remove(color) } else { checkedColors.insert(color) }
        defaults.set(Array(checkedColors), forKey: Keys.checked)
    }

    func clearColors() {
        checkedColors = []
        defaults.set([String](), forKey: Keys.checked)
    }

    func rememberColor(_ color: String) {
        guard !Self.defaultColors.contains(color) else { return }
        customColors = Array(([color] + customColors.filter { $0 != color }).prefix(8))
        defaults.set(customColors, forKey: Keys.custom)
    }

    /// Web版 localStorage の予定JSON(配列)を読み込む。同じIDは上書き。失敗時は何も変更しない
    @discardableResult
    func importJSON(_ data: Data) throws -> Int {
        let list = try JSONDecoder().decode([ShiftEvent].self, from: data)
        for ev in list {
            if let i = events.firstIndex(where: { $0.id == ev.id }) { events[i] = ev } else { events.append(ev) }
        }
        persist()
        return list.count
    }

    private func persist() {
        guard let data = try? JSONEncoder().encode(events) else { return }
        try? data.write(to: fileURL, options: .atomic)
    }
}
```

```swift
import SwiftUI
import UIKit

extension Color {
    init(hex: String) {
        var s = hex.trimmingCharacters(in: CharacterSet(charactersIn: "#"))
        if s.count == 3 { s = s.map { "\($0)\($0)" }.joined() }
        var v: UInt64 = 0
        Scanner(string: s).scanHexInt64(&v)
        self.init(red: Double((v >> 16) & 0xFF) / 255, green: Double((v >> 8) & 0xFF) / 255, blue: Double(v & 0xFF) / 255)
    }

    func toHex() -> String {
        var r: CGFloat = 0, g: CGFloat = 0, b: CGFloat = 0, a: CGFloat = 0
        UIColor(self).getRed(&r, green: &g, blue: &b, alpha: &a)
        return String(format: "#%02x%02x%02x", Int((r * 255).rounded()), Int((g * 255).rounded()), Int((b * 255).rounded()))
    }
}

enum Theme {
    static let accent = Color(hex: "#4f46e5")
    static let saturday = Color(hex: "#3b82f6")
    static let sunday = Color(hex: "#ef4444")
}

extension DayType {
    var color: Color {
        switch self {
        case .normal: return Color.secondary
        case .saturday: return Theme.saturday
        case .sunday, .holiday: return Theme.sunday
        }
    }
```

```swift
struct WeekView: View {

        let allDayRows = SpanLayout.rows(events: store.visibleEvents.filter(\.allDay), weekCells: days)

                         leading: timeW, onSelect: onSelectEvent)
            }
            .frame(height: allDayRows.isEmpty ? 0 : CGFloat(allDayRows.count) * (barH + 2) + 4)
            Divider()
            grid(days)
        }
    }

    private func header(_ ds: String) -> some View {
        let holiday = holidays.name(on: ds)
        let type = Rules.dayType(ds, isHoliday: holiday != nil)
        let isToday = ds == Rules.today()
        return VStack(spacing: 1) {
            Text(Rules.weekdayLabels(startMonday: false)[Rules.weekday(ds)])
                .font(.system(size: 9))
                .foregroundStyle(type.color)
            Text("\(Rules.ymd(ds).day)")
                .font(.system(size: 13, weight: .semibold))
                .foregroundStyle(isToday ? Color.white : (type == .normal ? Color.primary : type.color))
                .frame(minWidth: 22, minHeight: 22)
                .background(isToday ? Theme.accent : Color.clear, in: Circle())
        }
        .frame(maxWidth: .infinity)
        .padding(.vertical, 4)
        .contentShape(Rectangle())
        .onTapGesture { onSelectDay(ds) }
    }

    private func grid(_ days: [String]) -> some View {
        GeometryReader { geo in
            let colW = (geo.size.width - timeW) / 7
            let H = geo.size.height
            ZStack(alignment: .topLeading) {
                ForEach(0..<24, id: \.self) { h in
                    Rectangle()
                        .fill(Color(.separator).opacity(h % 3 == 0 ? 0.7 : 0.25))
                        .frame(width: geo.size.width - timeW, height: 0.5)
                        .offset(x: timeW, y: H * CGFloat(h) / 24)
                    if h % 3 == 0 {
                        Text("\(h)")
                            .font(.system(size: 8))
                            .foregroundStyle(.secondary)
                            .frame(width: timeW - 4, alignment: .trailing)
                            .offset(y: H * CGFloat(h) / 24 - 5)
                    }
                }
                ForEach(0..<7, id: \.self) { i in
                    column(days[i], colW: colW, H: H)
                        .frame(width: colW, height: H, alignment: .topLeading)
                        .overlay(alignment: .leading) {
                            Rectangle().fill(Color(.separator).opacity(0.5)).frame(width: 0.5)
                        }
                        .offset(x: timeW + colW * CGFloat(i))
                }
            }
        }
    }

    private func column(_ ds: String, colW: CGFloat, H: CGFloat) -> some View {
        ZStack(alignment: .topLeading) {
            Color.clear.contentShape(Rectangle()).onTapGesture { onSelectDay(ds) }
            ForEach(store.visibleEvents.filter { !$0.allDay }) { ev in
                if let seg = Rules.timedSegment(ev, on: ds) {
                    let h = max(14, H * seg.height)
                    block(ev, h: h)
                        .frame(width: colW - 2, height: h)
                        .offset(x: 1, y: H * seg.top)
                        .onTapGesture { onSelectEvent(ev) }
                }
            }
        }
    }

    private func block(_ ev: ShiftEvent, h: CGFloat) -> some View {
        VStack(alignment: .leading, spacing: 0) {

                Text(Rules.timeLabel(ev)).font(.system(size: 7)).lineLimit(1)
            }

        .foregroundStyle(.white)
        .padding(2)
        .frame(maxWidth: .infinity, maxHeight: .infinity, alignment: .topLeading)
        .background(Color(hex: ev.color).opacity(0.9), in: RoundedRectangle(cornerRadius: 3))
    }
}
```

```swift
import SwiftUI

struct EventEditor: View {
    @Environment(Store.self) private var store
    @Environment(\.dismiss) private var dismiss

    let original: ShiftEvent
    let isNew: Bool

    @State private var title: String
    @State private var memo: String
    @State private var color: String
    @State private var start: Date
    @State private var end: Date
    @State private var allDay: Bool
    @State private var startTime: String
    @State private var endTime: String
    @State private var breakText: String
    @State private var confirmDelete = false

    init(original: ShiftEvent, isNew: Bool) {
        self.original = original
        self.isNew = isNew
        _title = State(initialValue: original.title)
        _memo = State(initialValue: original.memo)
        _color = State(initialValue: original.color)
        _start = State(initialValue: Rules.parse(original.startDate ?? "") ?? Date())
        _end = State(initialValue: Rules.parse(original.endDate ?? original.startDate ?? "") ?? Date())
        _allDay = State(initialValue: original.allDay)
        _startTime = State(initialValue: original.startTime ?? "09:00")
        _endTime = State(initialValue: original.endTime ?? "18:00")
        _breakText = State(initialValue: Rules.breakOverride(of: original).map(String.init) ?? "")
    }

    private var canSave: Bool { !title.trimmingCharacters(in: .whitespaces).isEmpty }
    private var total: Int? { allDay ? nil : Rules.totalMinutes(start: startTime, end: endTime) }
    private var defaultBreak: Int { total.map { Rules.defaultBreak($0) } ?? 0 }
    private var hours: Double? {
        Rules.workHours(start: startTime, end: endTime, allDay: allDay, breakMinutes: Int(breakText))
    }

    var body: some View {
        NavigationStack {
            Form {
                Section {
                    TextField("タイトル", text: $title)
                    ColorRow(color: $color, recent: store.customColors)
                }
                Section {
                    DatePicker("開始日", selection: $start, displayedComponents: .date)
                    DatePicker("終了日", selection: $end, in: start..., displayedComponents: .date)
                    Toggle("終日", isOn: $allDay)
                    if !allDay { timeRows }
                }
                Section {
                    TextField("メモ(任意)", text: $memo, axis: .vertical).lineLimit(2...5)
                }
                Section {
                    Button("複製保存") { commit(duplicate: true) }.disabled(!canSave)
                    if !isNew {
                        Button("削除", role: .destructive) { confirmDelete = true }
                    }
                }
            }
            .navigationTitle(isNew ? "予定を追加" : "予定を編集")
            .navigationBarTitleDisplayMode(.inline)
            .toolbar {
                ToolbarItem(placement: .cancellationAction) { Button("キャンセル") { dismiss() } }
                ToolbarItem(placement: .confirmationAction) {
                    Button(isNew ? "追加" : "保存") { commit(duplicate: false) }.disabled(!canSave)
                }
            }
            .onChange(of: start) { _, new in if end < new { end = new } }
            .onChange(of: startTime) { breakText = "" }
            .onChange(of: endTime) { breakText = "" }
            .confirmationDialog("この予定を削除しますか?", isPresented: $confirmDelete, titleVisibility: .visible) {
                Button("削除", role: .destructive) {
                    store.delete(id: original.id)
                    dismiss()
                }
            }
        }
        .presentationDetents([.medium, .large])
    }

    @ViewBuilder
    private var timeRows: some View {
        Picker("開始", selection: $startTime) {
            ForEach(Rules.timeOptions, id: \.self) { Text($0).tag($0) }
        }
        Picker("終了", selection: $endTime) {
            ForEach(Rules.timeOptions, id: \.self) { Text($0).tag($0) }
        }
        if endTime <= startTime {
            Label("翌日またぎ → 終了日を自動で翌日にします", systemImage: "moon.stars")
                .font(.footnote)
                .foregroundStyle(.secondary)
        }
        HStack {
            Text("休憩")
            Spacer()
            TextField("\(defaultBreak)", text: $breakText)
                .keyboardType(.numberPad)
                .multilineTextAlignment(.trailing)
                .frame(width: 60)
            Text("分")
        }
        if let hours {
            LabeledContent("実労働", value: Rules.formatHours(hours))
        }
    }

    private func build() -> ShiftEvent {
        var e = original
        let st = allDay ? nil : startTime
        let et = allDay ? nil : endTime
        let sd = Rules.format(start)
        e.title = title.trimmingCharacters(in: .whitespaces)
        e.memo = memo
        e.color = color
        e.startDate = sd
        e.endDate = Rules.resolveEndDate(start: sd, end: Rules.format(end), allDay: allDay, startTime: st, endTime: et)
        e.placed = true
        e.allDay = allDay
        e.startTime = st
        e.endTime = et
        e.workHours = Rules.workHours(start: st, end: et, allDay: allDay, breakMinutes: Int(breakText))
        return e
    }

    private func commit(duplicate: Bool) {
```

```swift
            }
            .fileImporter(isPresented: $importing, allowedContentTypes: [.json]) { handleFile($0) }
            .alert("読み込み", isPresented: Binding(get: { message != nil }, set: { if !$0 { message = nil } })) {

            } message: {

            }

    private func importFromClipboard() {
        guard let text = UIPasteboard.general.string else {
            message = "クリップボードに文字列がありません"
            return
        }
        run { try store.importJSON(Data(text.utf8)) }
    }

    private func handleFile(_ result: Result<URL, Error>) {
        run {
            let url = try result.get()
            let scoped = url.startAccessingSecurityScopedResource()
            defer { if scoped { url.stopAccessingSecurityScopedResource() } }
            return try store.importJSON(Data(contentsOf: url))
        }
    }

    private func run(_ work: () throws -> Int) {
        do {
            message = "\(try work())件の予定を読み込みました"
        } catch {
            message = "読み込めませんでした: \(error.localizedDescription)"
        }
    }
}
```

```swift
import SwiftUI

enum CalendarMode: String { case week, month }

struct EditorTarget: Identifiable {
    let event: ShiftEvent
    let isNew: Bool
    var id: String { event.id }
}

struct RootView: View {
    @Environment(Store.self) private var store
    @Environment(Holidays.self) private var holidays
    @AppStorage("mode") private var mode: CalendarMode = .week
    @AppStorage("startMonday") private var startMonday = false
    @AppStorage("showSummary") private var showSummary = 0   // 0 なし / 1 週計 / 2 月計

    @State private var anchor = Rules.today()
    @State private var editing: EditorTarget?
    @State private var showFilter = false
    @State private var showSettings = false

    var body: some View {
        VStack(spacing: 0) {
            header
            Divider()
            Group {
                switch mode {
                case .month:
                    MonthView(anchor: anchor, startMonday: startMonday, onSelectDay: addEvent, onSelectEvent: edit)
                case .week:
                    WeekView(anchor: anchor, startMonday: startMonday, onSelectDay: addEvent, onSelectEvent: edit)
                }
            }
            .transition(.opacity)
            .contentShape(Rectangle())
            .simultaneousGesture(swipe)
            if showSummary > 0 {
                SummaryBar(label: summary.label, totals: summary.totals)
            }
        }
        .task(id: Rules.ymd(anchor).year) { await holidays.load(around: Rules.ymd(anchor).year) }
        .sheet(item: $editing) { EventEditor(original: $0.event, isNew: $0.isNew) }
        .sheet(isPresented: $showFilter) { FilterSheet() }
        .sheet(isPresented: $showSettings) { SettingsView() }
    }

    private var header: some View {
        VStack(spacing: 8) {
            HStack(spacing: 4) {
                Button { step(-1) } label: { Image(systemName: "chevron.left") }
                Text(title).font(.headline).frame(maxWidth: .infinity)
                Button { step(1) } label: { Image(systemName: "chevron.right") }
            }
            HStack(spacing: 12) {
                Picker("表示", selection: $mode.animation(.snappy)) {
                    Text("週").tag(CalendarMode.week)
                    Text("月").tag(CalendarMode.month)
                }
                .pickerStyle(.segmented)
                .frame(width: 120)
                Button("今日") { withAnimation(.snappy) { anchor = Rules.today() } }
                    .font(.subheadline)
                Spacer()
                Button { showSummary = (showSummary + 1) % 3 } label: {
                    Text(["集計", "週計", "月計"][showSummary])
                        .font(.caption.weight(.medium))
                        .padding(.horizontal, 8)
                        .padding(.vertical, 4)
                        .overlay(Capsule().stroke(showSummary > 0 ? Color.accentColor : Color.secondary.opacity(0.4)))
                }
                Button { showFilter = true } label: {
                    Image(systemName: store.checkedColors.isEmpty
                          ? "line.3.horizontal.decrease.circle" : "line.3.horizontal.decrease.circle.fill")
                }
                Button { showSettings = true } label: { Image(systemName: "gearshape") }
                Button { addEvent(on: anchor) } label: { Image(systemName: "plus.circle.fill") }
            }
        }
        .padding(.horizontal, 12)
        .padding(.vertical, 8)
    }

    private var title: String {
        if mode == .month {
            let p = Rules.ymd(anchor)
            return "\(p.year)年\(p.month)月"
        }
        let s = Rules.weekStart(of: anchor, startMonday: startMonday)
        let sp = Rules.ymd(s), ep = Rules.ymd(Rules.addDays(s, 6))
        return sp.month == ep.month
            ? "\(sp.year)年\(sp.month)月\(sp.day)〜\(ep.day)日"
            : "\(sp.month)月\(sp.day)日〜\(ep.month)月\(ep.day)日"
    }

    private var summary: (label: String, totals: [ColorTotal]) {
        if showSummary == 2 {
            let b = Rules.monthBounds(of: anchor)
            return ("月計", store.totals(from: b.first, to: b.last))
        }
        let s = Rules.weekStart(of: anchor, startMonday: startMonday)
        return ("週計", store.totals(from: s, to: Rules.addDays(s, 6)))
    }

    private var swipe: some Gesture {
        DragGesture(minimumDistance: 30).onEnded { v in
            guard abs(v.translation.width) > 60, abs(v.translation.width) > abs(v.translation.height) * 1.5 else { return }
            step(v.translation.width < 0 ? 1 : -1)
        }
    }

    private func step(_ n: Int) {
        withAnimation(.snappy) {
            anchor = mode == .month ? Rules.addMonths(anchor, n) : Rules.addDays(anchor, 7 * n)
        }
        UIImpactFeedbackGenerator(style: .light).impactOccurred()
    }

    private func addEvent(on date: String) {
        editing = EditorTarget(event: .new(on: date), isNew: true)
    }

    private func edit(_ event: ShiftEvent) {
        editing = EditorTarget(event: event, isNew: false)
    }
}
```
